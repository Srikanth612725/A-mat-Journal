def create_exact_ashaped_geometry():
    """
    Create EXACT A-shaped mat geometry from Liu et al. (2022)
    Area: 68.4 m^2
    """
    vertices = np.array([
        [5.0, 0.0],    # Bow tip
        [6.5, 2.0],    # Bow right
        [7.5, 5.0],    # Right forward
        [7.5, 9.0],    # Right back
        [6.25, 10.0],  # Stern right
        [5.0, 10.0],   # Stern center
        [3.75, 10.0],  # Stern left
        [2.5, 9.0],    # Left back
        [2.5, 5.0],    # Left forward
        [3.5, 2.0],    # Bow left
        [5.0, 0.0],    # Close
    ])

    perforation = np.array([
        [3.75, 1.5], [6.25, 1.5], [6.25, 4.0], [3.75, 4.0],
    ])

    wellhead_groove = np.array([
        [4.0, 8.0], [6.0, 8.0], [6.0, 10.0], [4.0, 10.0],
    ])

    return {
        'main_boundary': vertices,
        'perforation': perforation,
        'wellhead_groove': wellhead_groove,
        'area': 68.4,
        'length': 10.0,
        'width': 10.0,
    }

def point_in_polygon(x, y, polygon):
    """Check if point is inside polygon"""
    path = Path(polygon)
    return path.contains_point((x, y))

def is_inside_ashaped(x_local, y_local, geometry):
    """
    Check if LOCAL coordinates are inside A-shape
    x_local, y_local should be in [0, 10] range
    """
    if not point_in_polygon(x_local, y_local, geometry['main_boundary']):
        return False
    if point_in_polygon(x_local, y_local, geometry['perforation']):
        return False
    if point_in_polygon(x_local, y_local, geometry['wellhead_groove']):
        return False
    return True

@dataclass
class MaterialPoint:
    x: float
    y: float
    vx: float
    vy: float
    mass: float
    volume: float
    volume0: float
    sxx: float
    syy: float
    sxy: float
    exx: float
    eyy: float
    exy: float
    material_id: int  # 0=soil, 1=foundation

    def principal_stresses(self):
        s_mean = (self.sxx + self.syy) / 2
        radius = np.sqrt(((self.sxx - self.syy) / 2)**2 + self.sxy**2)
        return s_mean + radius, s_mean - radius

    def von_mises_stress(self):
        return np.sqrt(0.5 * ((self.sxx - self.syy)**2 +
                             self.sxx**2 + self.syy**2 + 6*self.sxy**2))

class MPM2D_AShaped:
    def __init__(self, domain_x, domain_y, nx, ny, su, E, nu, rho, geometry):
        # Domain
        self.x_min, self.x_max = domain_x
        self.y_min, self.y_max = domain_y
        self.Lx = self.x_max - self.x_min
        self.Ly = self.y_max - self.y_min

        # Grid
        self.nx = nx
        self.ny = ny
        self.dx = self.Lx / nx
        self.dy = self.Ly / ny
        self.nnx = nx + 1
        self.nny = ny + 1

        # Material
        self.su = su
        self.E = E
        self.nu = nu
        self.rho = rho
        self.G = E / (2 * (1 + nu))
        self.K = E / (3 * (1 - 2*nu))

        # Geometry
        self.geometry = geometry

        # Particles
        self.particles: List[MaterialPoint] = []
        self.foundation_indices = []

        # State
        self.foundation_y0 = None
        self.foundation_velocity = 0
        self.time = 0
        self.step_count = 0

        # Grid arrays
        self.reset_grid()

        print(f"MPM Solver initialized")
        print(f"  Domain: {self.Lx}m x {self.Ly}m")
        print(f"  Grid: {nx}x{ny} cells, {self.dx:.3f}m x {self.dy:.3f}m")

    def reset_grid(self):
        n = self.nnx * self.nny
        self.grid_mass = np.zeros(n)
        self.grid_vx = np.zeros(n)
        self.grid_vy = np.zeros(n)
        self.grid_fx = np.zeros(n)
        self.grid_fy = np.zeros(n)

    def node_index(self, i, j):
        return i * self.nny + j

    def add_soil_block(self, x_range, y_range, ppc=4):
        """Add soil particles"""
        x_min, x_max = x_range
        y_min, y_max = y_range

        ppc_1d = int(np.sqrt(ppc))
        dx_p = self.dx / ppc_1d
        dy_p = self.dy / ppc_1d
        V_p = self.dx * self.dy / ppc
        m_p = self.rho * V_p

        count = 0
        for x in np.arange(x_min + dx_p/2, x_max, dx_p):
            for y in np.arange(y_min + dy_p/2, y_max, dy_p):
                depth = max(0, y_max - y)
                syy_init = -self.rho * 9.81 * depth
                sxx_init = 0.5 * syy_init

                self.particles.append(MaterialPoint(
                    x=x, y=y, vx=0, vy=0,
                    mass=m_p, volume=V_p, volume0=V_p,
                    sxx=sxx_init, syy=syy_init, sxy=0,
                    exx=0, eyy=0, exy=0, material_id=0
                ))
                count += 1

        print(f"Added {count} soil particles (V={V_p:.6f} m^2, m={m_p:.4f} kg)")

    def add_ashaped_foundation(self, y_base, foundation_density=2500):
        """
        Add A-shaped foundation particles

        CRITICAL FIX: Properly center foundation in domain
        """
        geom = self.geometry

        # Particle spacing
        ppc = 4
        ppc_1d = int(np.sqrt(ppc))
        dx_p = self.dx / ppc_1d
        dy_p = self.dy / ppc_1d

        # Volume and mass per particle
        V_p = dx_p * dy_p
        mat_thickness = 0.5  # Out-of-plane thickness
        m_p = foundation_density * V_p * mat_thickness

        # CRITICAL: Center foundation in domain
        # Foundation local coords: x in [2.5, 7.5], y in [0, 10]
        # Domain center: x = (x_min + x_max) / 2
        domain_center_x = (self.x_min + self.x_max) / 2
        foundation_center_x = 5.0  # Center of A-shape in local coords
        x_offset = domain_center_x - foundation_center_x  # Shift to center

        count = 0
        start_idx = len(self.particles)

        # Sample across local foundation coordinates
        for x_local in np.arange(dx_p/2, 10.0, dx_p):
            for y_local in np.arange(dy_p/2, 10.0, dy_p):
                if is_inside_ashaped(x_local, y_local, geom):
                    # Transform to global coordinates
                    x_global = x_local + x_offset
                    y_global = y_local + y_base

                    self.particles.append(MaterialPoint(
                        x=x_global, y=y_global, vx=0, vy=0,
                        mass=m_p, volume=V_p, volume0=V_p,
                        sxx=0, syy=0, sxy=0,
                        exx=0, eyy=0, exy=0, material_id=1
                    ))
                    count += 1

        self.foundation_indices = list(range(start_idx, len(self.particles)))

        if self.foundation_indices:
            fy = [self.particles[i].y for i in self.foundation_indices]
            fx = [self.particles[i].x for i in self.foundation_indices]
            self.foundation_y0 = np.mean(fy)

            print(f"Added {count} foundation particles")
            print(f"  X range: [{min(fx):.1f}, {max(fx):.1f}] m")
            print(f"  Y range: [{min(fy):.1f}, {max(fy):.1f}] m")
            print(f"  Centroid: ({np.mean(fx):.1f}, {self.foundation_y0:.1f}) m")
            print(f"  Total mass: {sum(self.particles[i].mass for i in self.foundation_indices):.0f} kg")
        else:
            print("ERROR: No foundation particles created!")
            self.foundation_y0 = y_base + 5

        self._visualize()

    def _visualize(self):
        """Visualize particle distribution"""
        soil = [(p.x, p.y) for p in self.particles if p.material_id == 0]
        found = [(p.x, p.y) for p in self.particles if p.material_id == 1]

        fig, ax = plt.subplots(figsize=(12, 10))

        if soil:
            sx, sy = zip(*soil)
            ax.scatter(sx, sy, s=0.5, c='brown', alpha=0.3, label=f'Soil ({len(soil)})')

        if found:
            fx, fy = zip(*found)
            ax.scatter(fx, fy, s=3, c='red', marker='s', label=f'Foundation ({len(found)})')

        ax.set_xlabel('X (m)')
        ax.set_ylabel('Y (m)')
        ax.set_title('Particle Distribution')
        ax.legend()
        ax.set_aspect('equal')
        ax.grid(True, alpha=0.3)
        plt.tight_layout()
        plt.savefig('particle_distribution.png', dpi=150)
        plt.close()
        print("Saved: particle_distribution.png")

    def shape_function(self, x, x_node, h):
        """Linear shape function"""
        xi = (x - x_node) / h
        if abs(xi) > 1:
            return 0, 0
        N = 1 - abs(xi)
        dN = -np.sign(xi) / h if xi != 0 else 0
        return N, dN

    def get_shape_functions(self, mp):
        """Get shape functions for material point"""
        i = int((mp.x - self.x_min) / self.dx)
        j = int((mp.y - self.y_min) / self.dy)
        i = max(0, min(i, self.nx - 1))
        j = max(0, min(j, self.ny - 1))

        nodes, N, dNdx, dNdy = [], [], [], []

        for di in range(2):
            for dj in range(2):
                ni, nj = i + di, j + dj
                if ni < self.nnx and nj < self.nny:
                    x_n = self.x_min + ni * self.dx
                    y_n = self.y_min + nj * self.dy
                    Nx, dNx = self.shape_function(mp.x, x_n, self.dx)
                    Ny, dNy = self.shape_function(mp.y, y_n, self.dy)
                    Nxy = Nx * Ny
                    if Nxy > 1e-12:
                        nodes.append(self.node_index(ni, nj))
                        N.append(Nxy)
                        dNdx.append(dNx * Ny)
                        dNdy.append(Nx * dNy)

        return nodes, np.array(N), np.array(dNdx), np.array(dNdy)

    def timestep(self):
        """CFL timestep"""
        c = np.sqrt(self.K / self.rho)
        return 0.3 * min(self.dx, self.dy) / c

    def tresca_return(self, mp):
        """Tresca plasticity for soil"""
        if mp.material_id != 0:
            return

        s1, s2 = mp.principal_stresses()
        f = abs(s1 - s2) - 2 * self.su

        if f > 0:
            s_mean = (s1 + s2) / 2
            diff = abs(s1 - s2)
            if diff < 1e-10:
                return

            scale = 2 * self.su / diff
            s1_new = s_mean + (s1 - s_mean) * scale
            s2_new = s_mean + (s2 - s_mean) * scale

            theta = 0.5 * np.arctan2(2*mp.sxy, mp.sxx - mp.syy)
            c2, s2t = np.cos(theta)**2, np.sin(theta)**2

            mp.sxx = s_mean + (s1_new - s_mean) * c2 + (s2_new - s_mean) * s2t
            mp.syy = s_mean + (s1_new - s_mean) * s2t + (s2_new - s_mean) * c2
            mp.sxy *= scale

    def mpm_step(self, dt):
        """Single MPM timestep"""
        self.reset_grid()

        # P2G: Mass and momentum
        for mp in self.particles:
            nodes, N, dNdx, dNdy = self.get_shape_functions(mp)
            for k, n in enumerate(nodes):
                self.grid_mass[n] += N[k] * mp.mass
                self.grid_vx[n] += N[k] * mp.mass * mp.vx
                self.grid_vy[n] += N[k] * mp.mass * mp.vy

        # Grid velocities
        active = self.grid_mass > 1e-12
        self.grid_vx[active] /= self.grid_mass[active]
        self.grid_vy[active] /= self.grid_mass[active]

        # Internal forces
        for mp in self.particles:
            nodes, N, dNdx, dNdy = self.get_shape_functions(mp)
            for k, n in enumerate(nodes):
                self.grid_fx[n] -= mp.volume * (mp.sxx * dNdx[k] + mp.sxy * dNdy[k])
                self.grid_fy[n] -= mp.volume * (mp.sxy * dNdx[k] + mp.syy * dNdy[k])

        # Gravity on soil
        for mp in self.particles:
            if mp.material_id == 0:
                nodes, N, _, _ = self.get_shape_functions(mp)
                for k, n in enumerate(nodes):
                    self.grid_fy[n] -= N[k] * mp.mass * 9.81

        # Update grid velocity
        self.grid_vx[active] += dt * self.grid_fx[active] / self.grid_mass[active]
        self.grid_vy[active] += dt * self.grid_fy[active] / self.grid_mass[active]

        # Boundary conditions
        for i in range(self.nnx):
            n = self.node_index(i, 0)
            self.grid_vx[n] = 0
            self.grid_vy[n] = 0

        for j in range(self.nny):
            self.grid_vx[self.node_index(0, j)] = 0
            self.grid_vx[self.node_index(self.nnx-1, j)] = 0

        # Foundation prescribed velocity
        for idx in self.foundation_indices:
            self.particles[idx].vx = 0
            self.particles[idx].vy = self.foundation_velocity

        # G2P: Update particles
        for mp in self.particles:
            if mp.material_id == 0:  # Soil
                nodes, N, dNdx, dNdy = self.get_shape_functions(mp)
                if not nodes:
                    continue

                mp.vx = sum(N[k] * self.grid_vx[n] for k, n in enumerate(nodes))
                mp.vy = sum(N[k] * self.grid_vy[n] for k, n in enumerate(nodes))

                # Strain rate
                dvx_dx = sum(self.grid_vx[n] * dNdx[k] for k, n in enumerate(nodes))
                dvy_dy = sum(self.grid_vy[n] * dNdy[k] for k, n in enumerate(nodes))
                dvx_dy = sum(self.grid_vx[n] * dNdy[k] for k, n in enumerate(nodes))
                dvy_dx = sum(self.grid_vy[n] * dNdx[k] for k, n in enumerate(nodes))

                dexx = dt * dvx_dx
                deyy = dt * dvy_dy
                dexy = dt * 0.5 * (dvx_dy + dvy_dx)

                mp.exx += dexx
                mp.eyy += deyy
                mp.exy += dexy

                # Volume update
                mp.volume *= (1 + dexx + deyy)

                # Stress update (plane strain)
                dv = dexx + deyy
                mp.sxx += self.K * dv + 2 * self.G * (dexx - dv/3)
                mp.syy += self.K * dv + 2 * self.G * (deyy - dv/3)
                mp.sxy += 2 * self.G * dexy

                # Plasticity
                self.tresca_return(mp)

            # Position update
            mp.x += dt * mp.vx
            mp.y += dt * mp.vy

        self.time += dt
        self.step_count += 1

    def calculate_reaction(self):
        """
        Calculate bearing capacity reaction force

        CORRECTED METHOD:
        Integrate contact pressure at foundation base
        """
        if not self.foundation_indices:
            return 0

        # Find foundation base (lowest y of foundation particles)
        found_y = [self.particles[i].y for i in self.foundation_indices]
        found_x = [self.particles[i].x for i in self.foundation_indices]
        foundation_base = min(found_y)
        foundation_x_min = min(found_x)
        foundation_x_max = max(found_x)

        # Find soil particles just below foundation
        contact_depth = 2 * self.dy

        reaction = 0
        contact_particles = 0

        for mp in self.particles:
            if mp.material_id == 0:  # Soil
                # Check if particle is below foundation and within its x-extent
                if (foundation_base - contact_depth < mp.y < foundation_base and
                    foundation_x_min - self.dx < mp.x < foundation_x_max + self.dx):

                    # Contact pressure contribution
                    # In 2D plane strain: force = stress × width
                    # Each particle represents a width of dx_p
                    ppc_1d = 2  # sqrt(4)
                    dx_p = self.dx / ppc_1d

                    # Vertical stress (compressive is negative, so use -syy)
                    contact_pressure = -mp.syy  # Make positive for compression

                    # Weight by proximity to foundation base
                    dist = foundation_base - mp.y
                    weight = max(0, 1 - dist / contact_depth)

                    reaction += contact_pressure * dx_p * weight
                    contact_particles += 1

        return reaction

    def run_test(self, rate=0.01, target=0.5, interval=0.02, max_steps=10000):
        """Run bearing capacity test"""
        print("\n" + "="*70)
        print("BEARING CAPACITY TEST")
        print("="*70)

        if not self.foundation_indices:
            print("ERROR: No foundation!")
            return [], [], []

        dt = self.timestep()
        print(f"Rate: {rate} m/s, Target: {target} m")
        print(f"Timestep: {dt:.6f} s")
        print(f"Expected steps: {int(target/rate/dt)}")

        self.foundation_velocity = -rate

        settlements, loads, times = [], [], []
        next_out = 0
        t0 = time.time()

        while self.step_count < max_steps:
            self.mpm_step(dt)

            # Settlement
            current_y = np.mean([self.particles[i].y for i in self.foundation_indices])
            settlement = self.foundation_y0 - current_y

            if self.time >= next_out:
                load = self.calculate_reaction() / 1000  # kN
                settlements.append(settlement)
                loads.append(load)
                times.append(self.time)
                next_out += interval

                if self.step_count % 200 == 0:
                    elapsed = time.time() - t0
                    print(f"Step {self.step_count:5d} | s={settlement:.4f}m | "
                          f"Q={load:.0f}kN | t={elapsed:.0f}s", end='\r')

            if settlement >= target:
                print(f"\nTarget reached at step {self.step_count}")
                break

        print(f"\nDone in {time.time()-t0:.0f}s")
        return np.array(settlements), np.array(loads), np.array(times)

def main():
    print("\n" + "="*70)
    print("STARTING SIMULATION")
    print("="*70)

    geometry = create_exact_ashaped_geometry()

    # Visualize geometry
    fig, ax = plt.subplots(figsize=(8, 8))
    poly = Polygon(geometry['main_boundary'], fill=False, edgecolor='blue', linewidth=2)
    ax.add_patch(poly)
    perf = Polygon(geometry['perforation'], fill=True, facecolor='gray', alpha=0.5)
    ax.add_patch(perf)
    groove = Polygon(geometry['wellhead_groove'], fill=True, facecolor='gray', alpha=0.5)
    ax.add_patch(groove)
    ax.set_xlim(-1, 11)
    ax.set_ylim(-1, 11)
    ax.set_aspect('equal')
    ax.set_title('A-Shaped Foundation Geometry')
    ax.grid(True, alpha=0.3)
    plt.savefig('geometry.png', dpi=150)
    plt.close()
    print("Saved: geometry.png")

    # Setup
    su = LIU_DATA['soil_strength_su']
    E = 500 * su
    nu = 0.495
    rho = LIU_DATA['soil_density']

    # Domain - 3x foundation width to avoid boundary effects
    domain_width = 30.0
    domain_height = 25.0
    soil_surface = 12.0

    # Grid
    nx, ny = 60, 50

    print(f"\nMaterial: su={su}Pa, E={E/1000}kPa, rho={rho}kg/m3")
    print(f"Domain: {domain_width}x{domain_height}m, Grid: {nx}x{ny}")

    # Create solver
    mpm = MPM2D_AShaped(
        domain_x=[0, domain_width],
        domain_y=[0, domain_height],
        nx=nx, ny=ny,
        su=su, E=E, nu=nu, rho=rho,
        geometry=geometry
    )

    # Add materials
    mpm.add_soil_block([0, domain_width], [0, soil_surface], ppc=4)
    mpm.add_ashaped_foundation(soil_surface, foundation_density=2500)

    total = len(mpm.particles)
    soil = sum(1 for p in mpm.particles if p.material_id == 0)
    found = len(mpm.foundation_indices)
    print(f"\nParticles: {total} total ({soil} soil, {found} foundation)")

    if found < 100:
        print(f"WARNING: Only {found} foundation particles - may be inaccurate")

    # Run test
    settlements, loads, times = mpm.run_test(
        rate=0.01,
        target=0.5,
        interval=0.02,
        max_steps=10000
    )

    if len(loads) == 0:
        print("ERROR: No results!")
        return None, [], []

    # Find ultimate load
    if len(loads) > 5:
        try:
            from scipy.signal import savgol_filter
            w = min(11, len(loads)//2*2+1)
            smooth = savgol_filter(loads, w, 3) if w >= 3 else loads
            V_ult = np.max(smooth)
        except:
            V_ult = np.max(loads)
    else:
        V_ult = np.max(loads) if loads else 0

    # Plot results
    fig, axes = plt.subplots(1, 2, figsize=(14, 5))

    ax1 = axes[0]
    ax1.plot(settlements, loads, 'b-', lw=2, label='MPM')
    ax1.axhline(LIU_DATA['ultimate_load_test'], color='r', ls='--', lw=2,
                label=f"Test ({LIU_DATA['ultimate_load_test']} kN)")
    ax1.axhline(LIU_DATA['ultimate_load_FEM'], color='g', ls=':', lw=2,
                label=f"FEM ({LIU_DATA['ultimate_load_FEM']} kN)")
    ax1.set_xlabel('Settlement (m)')
    ax1.set_ylabel('Load (kN)')
    ax1.set_title('Load-Settlement Curve')
    ax1.legend()
    ax1.grid(True, alpha=0.3)

    ax2 = axes[1]
    methods = ['Test', 'FEM', 'MPM']
    values = [LIU_DATA['ultimate_load_test'], LIU_DATA['ultimate_load_FEM'], V_ult]
    colors = ['red', 'green', 'blue']
    bars = ax2.bar(methods, values, color=colors, alpha=0.7)
    for bar, v in zip(bars, values):
        ax2.text(bar.get_x() + bar.get_width()/2, bar.get_height(),
                f'{v:.0f}', ha='center', va='bottom', fontweight='bold')
    ax2.set_ylabel('Ultimate Load (kN)')
    ax2.set_title('Comparison')
    ax2.grid(True, alpha=0.3, axis='y')

    plt.tight_layout()
    plt.savefig('results.png', dpi=150)
    plt.show()

    # Summary
    print("\n" + "="*70)
    print("RESULTS")
    print("="*70)
    error = abs(V_ult - LIU_DATA['ultimate_load_test']) / LIU_DATA['ultimate_load_test'] * 100
    print(f"Test:  {LIU_DATA['ultimate_load_test']} kN")
